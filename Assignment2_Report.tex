\section{Problem 1}

\subsection{Problem 1A: Top-50 Document Frequency Terms}
\textbf{Goal:} compute document frequency and extract top 50 terms.

\textbf{Implementation summary:}
\begin{itemize}[leftmargin=*]
  \item Mapper emits term and document id.
  \item Reducer counts unique documents per term (DF).
  \item Final output is post-processed to produce \texttt{top50\_1a.tsv}.
\end{itemize}

\textbf{Output artifacts:}
\begin{itemize}[leftmargin=*]
  \item \texttt{downloads/output\_1a/part-r-00000} (50 lines)

\subsection{Problem 1B: Co-occurrence Pairs (Baseline)}
\textbf{Goal:} compute co-occurrence counts for top words using pair representation for window sizes $d \in \{1,2,3,4\}$.

\textbf{Implementation summary:}
\begin{itemize}[leftmargin=*]
  \item Mapper tokenizes each line and emits ordered pairs \texttt{(w\_i,w\_j)} within distance $d$.
  \item Reducer sums all pair counts.
\end{itemize}

\section{Problem 2}

\subsection{Problem 2A: Document Frequency over Full Vocabulary}
\textbf{Goal:} compute term document frequencies on full corpus (with stopword filtering and stemming), then generate top-100 DF file.

\textbf{Implementation summary:}
\begin{itemize}[leftmargin=*]
  \item Stopwords loaded from distributed cache \texttt{stopwords.txt}.
  \item Porter stemming applied.
  \item Document-frequency counting done with unique-term-per-document logic.
  \item Top 100 terms written to \texttt{top100\_df.tsv} for downstream use.
\end{itemize}

\subsection{Problem 2B: TF-DF Scoring using Top-100 DF Cache}
\textbf{Goal:} compute TF-DF style score per (document, term) for terms in \texttt{top100\_df.tsv}.

\textbf{Implementation summary:}
\begin{itemize}[leftmargin=*]
  \item Mapper loads \texttt{top100\_df.tsv} from distributed cache.
  \item Mapper computes per-document term frequencies for eligible terms.
  \item Reducer computes
  \[
  \text{score}(t,d) = tf(t,d) \cdot \ln\left(\frac{10000}{df(t)} + 1\right)
  \]
  and emits \texttt{doc\_id, term, score}.
\end{itemize}

  \item \texttt{downloads/output\_2b/part-r-00000}: 612,083 lines
